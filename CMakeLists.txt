cmake_minimum_required(VERSION 3.15...3.26)
project(pystl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
pybind11_add_module(_pystl src/pystl.cpp)
set_target_properties(_pystl PROPERTIES OUTPUT_NAME "pystl")
if(MSVC)
    target_compile_options(_pystl PRIVATE /O2 /arch:AVX2 /openmp)
else()
    target_compile_options(_pystl PRIVATE -O3 -march=native -fopenmp)
endif()
target_link_libraries(_pystl PRIVATE OpenMP::OpenMP_CXX)
install(TARGETS _pystl DESTINATION .)
